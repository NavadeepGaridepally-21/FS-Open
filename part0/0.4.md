sequenceDiagram
    participant browser
    participant server

    browser->>server: POST /exampleapp/new_note (note data)
    activate server

    %% Server saves the new note
    server-->>browser: Redirect response (status code + Location header)
    deactivate server

    %% Browser follows the redirect with a GET request
    browser->>server: GET /exampleapp/notes
    activate server
    server-->>browser: HTML document
    deactivate server

    browser->>server: GET /exampleapp/main.css
    activate server
    server-->>browser: CSS file
    deactivate server

    browser->>server: GET /exampleapp/main.js
    activate server
    server-->>browser: JavaScript file
    deactivate server

    %% JavaScript requests the notes data
    browser->>server: GET /exampleapp/data.json
    activate server
    server-->>browser: JSON containing notes
    deactivate server
